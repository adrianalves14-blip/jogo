const canvas = document.getElementById('gameCanvas'); const ctx = canvas.getContext('2d'); const scoreElement = document.getElementById('score'); canvas.width = window.innerWidth; canvas.height = window.innerHeight; let score = 0; let gameActive = true; const player = { x: canvas.width / 2, y: canvas.height / 2, radius: 15, color: '#3498db', speed: 5 // Velocidade de movimento }; // Objeto para monitorar as teclas pressionadas const keys = { w: false, a: false, s: false, d: false }; const projectiles = []; const enemies = []; // Listeners para o teclado window.addEventListener('keydown', (e) => { const key = e.key.toLowerCase(); if (keys.hasOwnProperty(key)) keys[key] = true; }); window.addEventListener('keyup', (e) => { const key = e.key.toLowerCase(); if (keys.hasOwnProperty(key)) keys[key] = false; }); // O mouse agora serve apenas para mirar e atirar let mouseX = 0; let mouseY = 0; window.addEventListener('mousemove', (e) => { mouseX = e.clientX; mouseY = e.clientY; }); window.addEventListener('mousedown', () => { if(!gameActive) return; const angle = Math.atan2(mouseY - player.y, mouseX - player.x); projectiles.push({ x: player.x, y: player.y, vx: Math.cos(angle) * 8, vy: Math.sin(angle) * 8 }); }); function spawnEnemy() { if (!gameActive) return; const radius = 15; let x, y; if (Math.random() < 0.5) { x = Math.random() < 0.5 ? 0 - radius : canvas.width + radius; y = Math.random() * canvas.height; } else { x = Math.random() * canvas.width; y = Math.random() < 0.5 ? 0 - radius : canvas.height + radius; } enemies.push({ x, y, speed: 2 }); } setInterval(spawnEnemy, 1000); function update() { if (!gameActive) return; // Movimentação WASD if (keys.w && player.y > 0) player.y -= player.speed; if (keys.s && player.y < canvas.height) player.y += player.speed; if (keys.a && player.x > 0) player.x -= player.speed; if (keys.d && player.x < canvas.width) player.x += player.speed; // Atualizar Projéteis projectiles.forEach((p, index) => { p.x += p.vx; p.y += p.vy; if (p.x < 0 || p.x > canvas.width || p.y < 0 || p.y > canvas.height) { projectiles.splice(index, 1); } }); // Atualizar Inimigos (Zumbis seguem o jogador) enemies.forEach((en, enIdx) => { const angle = Math.atan2(player.y - en.y, player.x - en.x); en.x += Math.cos(angle) * en.speed; en.y += Math.sin(angle) * en.speed; // Colisão com Jogador if (Math.hypot(player.x - en.x, player.y - en.y) < 25) { gameActive = false; alert("Zumbis te pegaram! Pontos: " + score); location.reload(); } // Colisão com Projétil projectiles.forEach((p, pIdx) => { if (Math.hypot(p.x - en.x, p.y - en.y) < 20) { enemies.splice(enIdx, 1); projectiles.splice(pIdx, 1); score += 10; scoreElement.innerText = score; } }); }); } function draw() { ctx.clearRect(0, 0, canvas.width, canvas.height); // Jogador ctx.fillStyle = player.color; ctx.fillRect(player.x - 15, player.y - 15, 30, 30); // Mira (uma linha sutil indicando onde você atira) ctx.strokeStyle = 'rgba(255,255,255,0.2)'; ctx.beginPath(); ctx.moveTo(player.x, player.y); ctx.lineTo(mouseX, mouseY); ctx.stroke(); // Projéteis ctx.fillStyle = '#f1c40f'; projectiles.forEach(p => { ctx.beginPath(); ctx.arc(p.x, p.y, 5, 0, Math.PI * 2); ctx.fill(); }); // Inimigos ctx.fillStyle = '#2ecc71'; enemies.forEach(en => { ctx.fillRect(en.x - 15, en.y - 15, 30, 30); }); requestAnimationFrame(() => { update(); draw(); }); } draw();
